cmake_minimum_required(VERSION 3.20)
project(RDKitPackageAcceptance LANGUAGES CXX)
enable_testing()
find_package(RDKit CONFIG REQUIRED)

function(package_test name)
  add_executable(${name} ${name}.cpp)
  target_compile_features(${name} PRIVATE cxx_std_20)
  target_link_libraries(${name} PRIVATE ${ARGN})
  add_test(NAME ${name} COMMAND ${name})
endfunction()

package_test(core RDKit::SmilesParse RDKit::SubstructMatch)
target_compile_definitions(core PRIVATE EXPECTED_RDKIT_VERSION="${EXPECTED_RDKIT_VERSION}")
package_test(fingerprints RDKit::Fingerprints RDKit::SmilesParse)
package_test(suppliers RDKit::FileParsers)
package_test(conformers RDKit::DistGeomHelpers RDKit::ForceFieldHelpers RDKit::EHTLib RDKit::SmilesParse)
package_test(drawing RDKit::MolDraw2D RDKit::SmilesParse)
if(TEST_CAIRO)
  package_test(cairo RDKit::MolDraw2D RDKit::SmilesParse)
endif()
if(TEST_INCHI)
  package_test(inchi RDKit::RDInchiLib RDKit::SmilesParse)
endif()
